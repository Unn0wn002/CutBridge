        return tr("package_status", {name: manifest.package_name || manifest.cut, fps: manifest.fps, frames: manifest.frames.count});
    }

    function aeHelpCopy(topic) {
        var prefix = "help_" + topic + "_";
        return {
            title: aeUiText(prefix + "title"),
            what: aeUiText(prefix + "what"),
            when: aeUiText(prefix + "when"),
            recommended: aeUiText(prefix + "recommended"),
            limits: aeUiText(prefix + "limits")
        };
    }

    function showContextHelp(topic) {
        var help = aeHelpCopy(topic);
        alert(
            "CutBridge — " + help.title + "\n\n" +
            aeUiText("help_what") + ": " + help.what + "\n\n" +
            aeUiText("help_when") + ": " + help.when + "\n\n" +
            aeUiText("help_recommended") + ": " + help.recommended + "\n\n" +
            aeUiText("help_limits") + ": " + help.limits
        );
    }

    function loadOnly(onLoaded, onError) {
        try {
            var m = chooseManifest();
            if (m && onLoaded) onLoaded(m);
            return m;
        } catch (e) {
            if (onError) onError(e);
            alertError(e.toString());
            return null;
        }
    }

    function buildUI(thisObj) {
        loadLocalePreference();
        var pal = (thisObj instanceof Panel) ? thisObj : new Window("palette", "CutBridge", undefined, {resizeable: true}); if (!pal) return pal;
        pal.orientation = "column"; pal.alignChildren = ["fill", "top"]; pal.spacing = 8; pal.margins = 12;

        var panelStatusKind = state.manifest ? "READY" : "WARNING";
        var panelStatusKey = state.manifest ? "status_package_loaded" : "status_load_prompt";

        var title = pal.add("statictext", undefined, "CutBridge");
        try { title.graphics.font = ScriptUI.newFont(title.graphics.font.name, "BOLD", 16); } catch (e) {}

        var localeRow = pal.add("group");
        localeRow.orientation = "row"; localeRow.alignChildren = ["left", "center"];
        var localeLabel = localeRow.add("statictext", undefined, tr("language_label"));
        var localeSelect = localeRow.add("dropdownlist", undefined, ["日本語", "English"]); localeSelect.selection = currentLocale === "JA" ? 0 : 1;

        var packagePanel = pal.add("panel", undefined, aeUiText("package_section"));
        packagePanel.orientation = "column"; packagePanel.alignChildren = ["fill", "top"]; packagePanel.spacing = 4; packagePanel.margins = 10;
        var packageName = packagePanel.add("statictext", undefined, ""); packageName.characters = 48;
        var packageMeta = packagePanel.add("statictext", undefined, packageStatusText(state.manifest)); packageMeta.characters = 48;
        var statusKind = packagePanel.add("statictext", undefined, ""); statusKind.characters = 48;
        var statusDetail = packagePanel.add("statictext", undefined, "", {multiline: true}); statusDetail.characters = 52; statusDetail.preferredSize.height = 34;

        var actionsPanel = pal.add("panel", undefined, aeUiText("actions_section"));
        actionsPanel.orientation = "column"; actionsPanel.alignChildren = ["fill", "top"]; actionsPanel.spacing = 5; actionsPanel.margins = 10;

        function addAction(buttonText, descriptionKey, helpTopic, handler) {
            var row = actionsPanel.add("group"); row.orientation = "row"; row.alignChildren = ["fill", "center"];
            var button = row.add("button", undefined, buttonText); button.alignment = ["fill", "center"];
            var helpButton = row.add("button", undefined, "?"); helpButton.preferredSize.width = 30;
            var description = actionsPanel.add("statictext", undefined, aeUiText(descriptionKey), {multiline: true});
            description.characters = 52; description.preferredSize.height = 34;
            button.onClick = handler;
            helpButton.onClick = function() { showContextHelp(helpTopic); };
            return {button: button, help: helpButton, description: description};
        }

        function statusLabelFor(kind) {
            if (kind === "ERROR") return aeUiText("status_error");
            if (kind === "READY") return aeUiText("status_ready");
            return aeUiText("status_warning");
        }

        function refreshPackagePanel() {
            var manifest = state.manifest;
            packagePanel.text = aeUiText("package_section");
            actionsPanel.text = aeUiText("actions_section");
            packageName.text = aeUiText("package_label") + ": " + (manifest ? (manifest.package_name || manifest.cut || aeUiText("none")) : aeUiText("none"));
            packageMeta.text = packageStatusText(manifest);
            statusKind.text = aeUiText("status_label") + ": " + statusLabelFor(panelStatusKind);
            statusDetail.text = aeUiText(panelStatusKey);
        }

        function setPanelStatus(kind, key) {
            panelStatusKind = kind;
            panelStatusKey = key;
            refreshPackagePanel();
        }

        var loadAction = addAction(tr("import_package"), "desc_load", "load", function() {
            loadOnly(
                function() { setPanelStatus("READY", "status_package_loaded"); },
                function() { setPanelStatus("ERROR", "status_action_failed"); }
            );
        });
        var buildAction = addAction(tr("build_comp"), "desc_build", "build", function() {
            var result = buildComp();
            if (result && result.success) setPanelStatus("READY", "status_build_complete");
            else if (result) setPanelStatus("ERROR", "status_action_failed");
        });
        var qcAction = addAction(tr("run_qc"), "desc_qc", "qc", function() {
            try {
                var result = runQC(), headline = result && result.report ? String(result.report.headline || "").toUpperCase() : "";
                if (headline.indexOf("ERROR") >= 0) setPanelStatus("ERROR", "status_qc_error");
                else if (headline.indexOf("WARNING") >= 0) setPanelStatus("WARNING", "status_qc_warning");
                else if (result && result.report) setPanelStatus("READY", "status_qc_pass");
            } catch (e) {
                setPanelStatus("ERROR", "status_action_failed");
                alertError(e.toString());
            }
        });
        var revisionAction = addAction(tr("update_revision"), "desc_revision", "revision", function() {
            var beforeVersion = state.manifest ? state.manifest.version : null;
            updateRevision();
            var afterVersion = state.manifest ? state.manifest.version : null;
            if (beforeVersion !== null && afterVersion !== null && beforeVersion !== afterVersion) setPanelStatus("READY", "status_revision_complete");
            else refreshPackagePanel();
        });

        var note = pal.add("statictext", undefined, tr("workflow_note", {version: CutBridgeContract.PRODUCT_VERSION}), {multiline: true}); note.preferredSize.height = 32;

        function refreshLocale() {
            localeSelect.selection = currentLocale === "JA" ? 0 : 1;
            localeLabel.text = tr("language_label");
            loadAction.button.text = tr("import_package"); loadAction.description.text = aeUiText("desc_load");
            buildAction.button.text = tr("build_comp"); buildAction.description.text = aeUiText("desc_build");
            qcAction.button.text = tr("run_qc"); qcAction.description.text = aeUiText("desc_qc");
            revisionAction.button.text = tr("update_revision"); revisionAction.description.text = aeUiText("desc_revision");
            note.text = tr("workflow_note", {version: CutBridgeContract.PRODUCT_VERSION});
            refreshPackagePanel();
            try { pal.layout.layout(true); } catch (layoutError) {}
        }

        localeSelect.onChange = function() {
            if (!localeSelect.selection) return;
            saveLocalePreference(localeSelect.selection.index === 0 ? "JA" : "EN");
            refreshLocale();
        };

        refreshPackagePanel();
        pal.onResizing = pal.onResize = function() { this.layout.resize(); };
        return pal;
    }
